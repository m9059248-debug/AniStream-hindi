🛠 Supabase SQL Schema & App Users Sync Helper

ऐप में लॉगिन करने वाले सभी यूज़र्स की डिटेल्स (Name, Email, Profile Photo, Last Login, My List) को Admin Panel में ऑटोमैटिक दिखाने और सभी नए फीचर्स चालू करने के लिए नीचे वाला SQL कॉपी करके Supabase SQL Editor में एक बार Run करें:

-- 1. Create App Users Profile Table & Auto-Sync Trigger from Supabase Auth CREATE TABLE IF NOT EXISTS public.user_profiles ( id uuid PRIMARY KEY, email text, full_name text, avatar_url text, my_list_count int DEFAULT 0, created_at timestamptz, last_sign_in_at timestamptz, updated_at timestamptz DEFAULT now() ); CREATE OR REPLACE FUNCTION public.sync_auth_user_to_profile() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_name text; v_avatar text; v_list_count int; BEGIN v_name := COALESCE( NULLIF(TRIM(NEW.raw_user_meta_data->>'full_name'), ''), NULLIF(TRIM(NEW.raw_user_meta_data->>'name'), ''), split_part(COALESCE(NEW.email, 'User'), '@', 1) ); v_avatar := COALESCE(NEW.raw_user_meta_data->>'avatar_url', ''); IF jsonb_typeof(NEW.raw_user_meta_data->'my_list') = 'array' THEN v_list_count := jsonb_array_length(NEW.raw_user_meta_data->'my_list'); ELSE v_list_count := 0; END IF; INSERT INTO public.user_profiles ( id, email, full_name, avatar_url, my_list_count, created_at, last_sign_in_at, updated_at ) VALUES ( NEW.id, NEW.email, v_name, v_avatar, v_list_count, NEW.created_at, NEW.last_sign_in_at, now() ) ON CONFLICT (id) DO UPDATE SET email = EXCLUDED.email, full_name = EXCLUDED.full_name, avatar_url = CASE WHEN EXCLUDED.avatar_url <> '' THEN EXCLUDED.avatar_url ELSE public.user_profiles.avatar_url END, my_list_count = EXCLUDED.my_list_count, last_sign_in_at = EXCLUDED.last_sign_in_at, updated_at = now(); RETURN NEW; END; $$; DROP TRIGGER IF EXISTS trg_sync_auth_user_to_profile ON auth.users; CREATE TRIGGER trg_sync_auth_user_to_profile AFTER INSERT OR UPDATE ON auth.users FOR EACH ROW EXECUTE FUNCTION public.sync_auth_user_to_profile(); INSERT INTO public.user_profiles ( id, email, full_name, avatar_url, my_list_count, created_at, last_sign_in_at, updated_at ) SELECT u.id, u.email, COALESCE( NULLIF(TRIM(u.raw_user_meta_data->>'full_name'), ''), NULLIF(TRIM(u.raw_user_meta_data->>'name'), ''), split_part(COALESCE(u.email, 'User'), '@', 1) ), COALESCE(u.raw_user_meta_data->>'avatar_url', ''), CASE WHEN jsonb_typeof(u.raw_user_meta_data->'my_list') = 'array' THEN jsonb_array_length(u.raw_user_meta_data->'my_list') ELSE 0 END, u.created_at, u.last_sign_in_at, now() FROM auth.users u ON CONFLICT (id) DO UPDATE SET email = EXCLUDED.email, full_name = EXCLUDED.full_name, avatar_url = EXCLUDED.avatar_url, my_list_count = EXCLUDED.my_list_count, last_sign_in_at = EXCLUDED.last_sign_in_at, updated_at = now(); GRANT USAGE ON SCHEMA public TO anon, authenticated; GRANT ALL ON TABLE public.user_profiles TO anon, authenticated, service_role; ALTER TABLE public.user_profiles ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Allow read access to user_profiles" ON public.user_profiles; CREATE POLICY "Allow read access to user_profiles" ON public.user_profiles FOR SELECT TO public, anon, authenticated USING (true); -- 2. Add Direct Download URL column to video_servers ALTER TABLE public.video_servers ADD COLUMN IF NOT EXISTS download_url text; -- 3. Add Popup & Announcement targeting/expiry columns to notifications ALTER TABLE public.notifications ADD COLUMN IF NOT EXISTS notif_type text DEFAULT 'homepage_popup'; ALTER TABLE public.notifications ADD COLUMN IF NOT EXISTS anime_id uuid; ALTER TABLE public.notifications ADD COLUMN IF NOT EXISTS episode_id uuid; ALTER TABLE public.notifications ADD COLUMN IF NOT EXISTS starts_at timestamptz; ALTER TABLE public.notifications ADD COLUMN IF NOT EXISTS expires_at timestamptz; -- 4. Create Anime Schedules table CREATE TABLE IF NOT EXISTS public.schedules ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), anime_id uuid REFERENCES public.anime(id) ON DELETE CASCADE, day_of_week text NOT NULL, release_time text DEFAULT '18:00', episode_number int DEFAULT 1, note text, created_at timestamptz DEFAULT now() ); -- 5. Create Site Settings table for Download Verification & 24h Unlock CREATE TABLE IF NOT EXISTS public.site_settings ( key text PRIMARY KEY, value jsonb NOT NULL, updated_at timestamptz DEFAULT now() ); -- 6. Create 24-Hour Verified Download Tokens table CREATE TABLE IF NOT EXISTS public.download_tokens ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_identifier text NOT NULL, episode_id uuid, quality text NOT NULL, ad_transaction_id text UNIQUE NOT NULL, verified boolean DEFAULT false, expires_at timestamptz NOT NULL, created_at timestamptz DEFAULT now() ); -- 7. Multi-language episode fix: ek episode = ek row (season + episode number se unique) -- Pehle admin panel me "Merge Duplicate Language Episodes" chalayein. Agar duplicates bache hon to ye step apne aap skip ho jayega. DO $$ BEGIN ALTER TABLE public.episodes DROP CONSTRAINT IF EXISTS episodes_anime_ep_lang_key; ALTER TABLE public.episodes ADD CONSTRAINT episodes_season_ep_key UNIQUE (season_id, episode_number); EXCEPTION WHEN OTHERS THEN RAISE NOTICE 'Episode key update skipped: %', SQLERRM; END $$; -- ===== Community Chat (App ke Chat tab ke liye) ===== -- Supabase Dashboard > SQL Editor me paste karke Run karein. CREATE TABLE IF NOT EXISTS public.chat_messages ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), user_id uuid NOT NULL DEFAULT auth.uid() REFERENCES auth.users(id) ON DELETE CASCADE, user_name text NOT NULL DEFAULT 'Anime Fan', username text, message text NOT NULL CHECK (char_length(message) BETWEEN 1 AND 500), created_at timestamptz NOT NULL DEFAULT now() ); CREATE INDEX IF NOT EXISTS chat_messages_created_idx ON public.chat_messages (created_at DESC); CREATE TABLE IF NOT EXISTS public.chat_bans ( user_id uuid PRIMARY KEY, user_name text, reason text, created_at timestamptz DEFAULT now() ); ALTER TABLE public.chat_messages ENABLE ROW LEVEL SECURITY; ALTER TABLE public.chat_bans ENABLE ROW LEVEL SECURITY; GRANT SELECT, INSERT, DELETE ON public.chat_messages TO authenticated; GRANT SELECT, INSERT, UPDATE, DELETE ON public.chat_bans TO authenticated; -- Sirf login kiye hue users hi chat padh sakte hain DROP POLICY IF EXISTS "chat read for logged in users" ON public.chat_messages; CREATE POLICY "chat read for logged in users" ON public.chat_messages FOR SELECT TO authenticated USING (true); -- Sirf login kiya hua user, apne hi naam se, aur block na hua ho to hi message bhej sakta hai DROP POLICY IF EXISTS "chat send for logged in users" ON public.chat_messages; CREATE POLICY "chat send for logged in users" ON public.chat_messages FOR INSERT TO authenticated WITH CHECK ( user_id = auth.uid() AND NOT EXISTS (SELECT 1 FROM public.chat_bans b WHERE b.user_id = auth.uid()) ); -- Messages sirf admin delete kar sakta hai DROP POLICY IF EXISTS "chat admin delete" ON public.chat_messages; CREATE POLICY "chat admin delete" ON public.chat_messages FOR DELETE TO authenticated USING (EXISTS (SELECT 1 FROM public.admin_users a WHERE a.user_id = auth.uid())); -- Ban list: user apni khud ki ban entry dekh sakta hai, admin sab manage kar sakta hai DROP POLICY IF EXISTS "chat bans read own or admin" ON public.chat_bans; CREATE POLICY "chat bans read own or admin" ON public.chat_bans FOR SELECT TO authenticated USING (user_id = auth.uid() OR EXISTS (SELECT 1 FROM public.admin_users a WHERE a.user_id = auth.uid())); DROP POLICY IF EXISTS "chat bans admin manage" ON public.chat_bans; CREATE POLICY "chat bans admin manage" ON public.chat_bans FOR ALL TO authenticated USING (EXISTS (SELECT 1 FROM public.admin_users a WHERE a.user_id = auth.uid())) WITH CHECK (EXISTS (SELECT 1 FROM public.admin_users a WHERE a.user_id = auth.uid())); -- Spam rokne ke liye 2 second ka gap, aur naam server se hi set hota hai (koi dusre ke naam se message nahi bhej sakta) CREATE OR REPLACE FUNCTION public.chat_prepare_message() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, auth AS $$ DECLARE u record; BEGIN IF EXISTS ( SELECT 1 FROM public.chat_messages m WHERE m.user_id = NEW.user_id AND m.created_at > now() - interval '2 seconds' ) THEN RAISE EXCEPTION 'Please wait 2 seconds between messages.'; END IF; SELECT COALESCE(NULLIF(raw_user_meta_data->>'full_name', ''), NULLIF(raw_user_meta_data->>'name', ''), split_part(email, '@', 1)) AS uname, NULLIF(raw_user_meta_data->>'username', '') AS handle INTO u FROM auth.users WHERE id = NEW.user_id; NEW.user_name := COALESCE(u.uname, 'Anime Fan'); NEW.username := u.handle; NEW.message := trim(NEW.message); RETURN NEW; END; $$; DROP TRIGGER IF EXISTS chat_prepare_message_trg ON public.chat_messages; CREATE TRIGGER chat_prepare_message_trg BEFORE INSERT ON public.chat_messages FOR EACH ROW EXECUTE FUNCTION public.chat_prepare_message(); NOTIFY pgrst, 'reload schema';
